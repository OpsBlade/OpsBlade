# The workflow name appears in alert subject lines. It defaults to the file name.
name: Example workflow
dryrun: false
debug: false
json: false
env: /home/user/.env

# Optional: send a loud WARNING (completed with on_fail: warn failures) or
# FATAL ERROR (aborted) alert. Slack, email, or both. Credentials come from
# the env file: SLACK_WEBHOOK[_suffix], SMTP_HOST, SMTP_PORT, SMTP_USER, SMTP_PASS.
notify:
  slack:
    env_suffix: _ALERTS   # optional; omit to use SLACK_WEBHOOK
    mention: channel      # optional; channel (default), here, or none
  email:
    to: [ops@example.com]
    from: "OpsBlade <opsblade@example.com>"   # a bare address also works
    subject_prefix: "[OpsBlade]"   # optional; applied to every email
    # transcript controls emailing the run's captured stdout and stderr:
    #   always: every run is emailed (COMPLETED, STOPPED, WARNING, FATAL ERROR) with the transcript (default)
    #   error:  only WARNING and FATAL ERROR alerts are emailed, with the transcript
    #   never:  only WARNING and FATAL ERROR alerts are emailed, without the transcript
    transcript: always

tasks:
    # Variables are a name value pair. The name must be a string. Enclosing string values in quotes
    # is recommended to ensure they are not interpreted as a number or boolean.
    # Task names are for human use only.
    # The task field is matched against the task registry and must be an exact match.
  - name: Set variables
    task: variables_set
    set:
    - name: something_wanted
      value: "yes"
    - name: "hint"
      value: "set debug to true if you'd like to see more output"

    # This is an example of a task that makes a call to an API
  - name: "Example call to an API"
    task: example

    # Filters are AWS EC2-compatible filters. The name is a string, the values are a list of strings. They are intended
    # to be passed in API calls.
    filters:
    - name: "whatever"
      values:
        - "true"

    # Select may be used to filter the results of an API call. It is up to the individual task to pass items
    # through the select filter as contextually appropriate.
    select:
      - field: paid
        value: "{{something_wanted}}"
        compare: equal

    # Fields is intended to select fields to be returned to the user. It is up to the individual task to pass
    # relevant items through the fields filter.
    fields:
      - id
      - tags.tag1

  # Every task accepts on_fail: warn | fatal | stop (default fatal).
  #   fatal: abort the workflow, exit 1, FATAL ERROR alert
  #   warn:  record a warning and continue, exit 0, WARNING alert at the end
  #   stop:  a normal condition; stop the workflow cleanly, exit 0, no alert
  # A notification that fails should not stop real work:
  - name: "Announce"
    task: slack_send
    subject: "Example workflow starting"
    body: "Hello"
    on_fail: warn
    skip: true  # Set to false to actually run this task

  # Example of loading variables with a custom error message (v0.1.11+)
  # The error_message field provides context when a task fails, useful for expected failures.
  # on_fail: stop makes the missing file a clean exit rather than an error.
  - name: "Load previous run results"
    task: "variables_load"
    filename: "/tmp/previous_run.json"
    error_message: "This is expected if this is the first run. The file is created after the first successful execution."
    on_fail: stop
    skip: true  # Set to false to actually run this task

  # jira_issue_check separates "not in the required state" from "could not check".
  #   on_mismatch: stop | warn | fatal (default stop) applies when the issue was fetched
  #                but its status or resolution does not match.
  #   on_fail:     applies to Jira errors (bad credentials, 401, 404, network) as usual.
  # With the defaults, a ticket that is not approved yet exits 0 quietly, while an
  # expired token exits 1 with a FATAL ERROR alert.
  - name: "Check that the change ticket is approved"
    task: jira_issue_check
    issue_id: "OPS-123"
    required_status: "Done / Closed"
    required_resolution: "Done"
    error_message: "The change ticket is not approved yet."
    on_mismatch: stop
    skip: true  # Set to false to actually run this task

  # Confirm the AWS credentials belong to the expected account before any AWS task
  # changes anything. Uses the same env/region/profile resolution as the other AWS tasks.
  #   on_mismatch: fatal | warn | stop (default fatal) applies when the account differs.
  #   on_fail:     applies to AWS errors (bad credentials, network) as usual.
  - name: "Confirm we are in the production AWS account"
    task: aws_account_check
    env: "aws-prod.env"
    account_id: "123456789012"
    skip: true  # Set to false to actually run this task

  # The save task is completely optional and allows filtering.
  # A common use case would be to store the output from a task for later processing. The file saved
  # with variables_saved can be loaded in another workflow with variables_load.
  - name: "Save all output from the tasks"
    task: "variables_save"
    filename: "/tmp/example1.json"

  # Fields can be selected. This is applied over the entire map of variables.
  # Note that since the task itself performed field filtering, only the fields that
  # were passed through the task exist as variables.
  - name: "Save mock_data only"
    task: "variables_save"
    fields:
      - mock_data
    filename: "/tmp/example2.json"

  # This time we only want the id and tag1 values
  - name: "Save variables, which includes the selected data"
    task: "variables_save"
    # This isn't the most useful, but it demonstrates the ability to filter data from within
    # a list.
    # "*" is used to indicate to iterate through all items in a slice and apply the next part of the
    # dot notation field name to the top level of the list. This example will result in it iterating
    # over all items in mock_data and returning a slice/list containing the id and tags.tag1 for each item.
    fields:
      - mock_data.*.id
      - mock_data.*.tags.tag1
    filename: "/tmp/example3.json"
